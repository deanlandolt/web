<script setup>
import { onMounted, ref } from 'vue'
const stamp = ref('404')
onMounted(() => {
  // tiny flourish: a slow blinking 404, retro terminal-style
  const labels = ['404', '   ']
  let i = 0
  if (!window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    setInterval(() => {
      i = (i + 1) % labels.length
      stamp.value = labels[i]
    }, 600)
  }
})
</script>

# {{ stamp }}
